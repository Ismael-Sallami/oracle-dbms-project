
\section{Primer requisito funcional}
\textbf{\textcolor{rojo}{\underline{RF4.1:}}} Crear usuario. \\

\textbf{\textcolor{rojo}{\underline{Descripción:}}} Permite a un visitante crear un usuario en el sistema, asignándole un nombre de usuario y una contraseña.
Opcionalmente, puede añadir una imagen de perfil y/o una breve biografía. \\

\textbf{\textcolor{rojo}{Entrada:}} Agente externo: visitante. Acción: solicitar inserción.
Requisito de datos de entrada \textcolor{azul}{RDE4.1}.\\ 
\textbf{\textcolor{rojo}{BD:}} Requisito de datos de escritura \textcolor{azul}{RDW4.1}. \\

\textbf{\textcolor{rojo}{Salida:}} Agente externo: visitante. Acción: confirmación resultado.
Requisito de datos de salida: ninguno.


\textcolor{azul}{\textbf{RDE4.1:}} Datos de entrada de crear usuario. \\
\hspace*{1cm} Nombre de usuario: Cadena de caracteres (20)\\
\hspace*{1cm} Correo electrónico: Cadena de caracteres (50)\\
\hspace*{1cm} Contraseña: Cadena de caracteres (20)\\
\hspace*{1cm} Imagen de perfil: Cadena de caracteres (255)\\
\hspace*{1cm} Biografía: Cadena de caracteres (100) \\

\textcolor{azul}{\textbf{RDW4.1:}} Datos almacenados del usuario. \\
\hspace*{1cm} Los mismos que RDE4.1 \\
\hspace*{1cm} ID de usuario: Cadena de caracteres (40) \\
\hspace*{1cm} Fecha de creación: Fecha \\

Requisito de datos de salida: ninguno. \\[12pt]

\hspace*{1cm} Contraseña: Cadena de caracteres (20)\\[8pt]

\textcolor{azul}{\textbf{RDW4.3:}} Datos eliminados del usuario. \\[4pt]
\hspace*{1cm} ID de usuario: Cadena de caracteres (40)\\
\hspace*{1cm} Demas datos del RDW4.1\\
\hspace*{1cm} Fecha de eliminación: Fecha \\

\section{Cuarto requisito funcional}
\textbf{\textcolor{rojo}{\underline{RF4.4:}}} Bloquear usuario. \\[6pt]

\textbf{\textcolor{rojo}{\underline{Descripción:}}} Permite a un usuario bloquear a otro usuario en base a su ID. Bloquear a un usuario implica que ninguno de los dos podrá interactuar con las publicaciones, comentarios o mensajes del otro dentro del sistema. \\[6pt]

\textbf{\textcolor{rojo}{\underline{Entrada:}}} Agente externo: usuario. Acción: bloquear a un usuario.
Requisito de datos de entrada \textcolor{azul}{RDE4.4}. \\[6pt]

\textbf{\textcolor{rojo}{\underline{BD:}}} Requisito de datos de escritura \textcolor{azul}{RDW4.4}.\\[6pt]

\textbf{\textcolor{rojo}{\underline{Salida:}}} Agente externo: usuario. Acción: confirmación resultado.
Requisito de datos de salida: ninguno.\\[12pt]


\textcolor{azul}{\textbf{RDE4.4:}} Datos de entrada de bloquear usuario. \\[4pt]
\hspace*{1cm} ID de usuario emisor: Cadena de caracteres (40)\\
\hspace*{1cm} ID de usuario a bloquear: Cadena de caracteres (40)\\
\hspace*{1cm} Fecha de bloqueo: Fecha\\[8pt]

\textcolor{azul}{\textbf{RDW4.4:}} Datos sobre el usuario bloqueado. \\[4pt]
\hspace*{1cm} Se añade el ID del usuario bloqueado a la lista de usuarios bloqueados del \hspace*{1cm} emisor.\\
\hspace*{1cm} Estado de relación: Cadena de caracteres (15) — (“bloqueado”)\\
\hspace*{1cm} Fecha de bloqueo: Fecha\\[8pt]

\textbf{\textcolor{verde}{Restricciones Semánticas:}} \\[6pt]

\textbf{\textcolor{verde}{\underline{RS4.1:}}} Un usuario no puede bloquearse a sí mismo. \textbf{RF:} RF4.4 \textbf{RD(s):} RDW4.4 \\ 
\hspace*{1cm} \textit{Descripción:} El sistema debe impedir que un usuario intente bloquear su propia cuenta, mostrando un mensaje de error si lo intenta. \\[4pt]

\textbf{\textcolor{verde}{\underline{RS4.2:}}} No se puede bloquear a un usuario inexistente o eliminado. \textbf{RF:} RF4.4 \textbf{RD(s):} RDW4.4 \\ 
\hspace*{1cm} \textit{Descripción:} Si el usuario objetivo no existe o su cuenta fue eliminada, la acción de bloqueo es rechazada. \\[4pt]

\textbf{\textcolor{verde}{\underline{RS4.3:}}} No se puede volver a bloquear un usuario ya bloqueado. \textbf{RF:} RF4.4 \textbf{RD(s):} RDW4.4. \\ 
\hspace*{1cm} \textit{Descripción:} El sistema verifica si el usuario ya se encuentra bloqueado y, de ser así, impide duplicar el bloqueo. \\[4pt]

\textbf{\textcolor{verde}{\underline{RS4.4:}}} El bloqueo impide la interacción con las publicaciones del usuario bloqueado. \textbf{RF:} RF4.4 \textbf{RD(s):} RDW4.4 \\ 
\hspace*{1cm} \textit{Descripción:} Una vez bloqueado un usuario, no se permite visualizar, comentar ni reaccionar a sus publicaciones. \\[4pt]

\textbf{\textcolor{verde}{\underline{RS4.5:}}} Solo quien realizó el bloqueo puede desbloquear. \textbf{RF:} RF4.4 \textbf{RD(s):} RDW4.4. \\ 
\hspace*{1cm} \textit{Descripción:} El desbloqueo solo puede ejecutarse por el usuario que originó el bloqueo. \\[4pt]

\textbf{\textcolor{verde}{\underline{RS4.6:}}} Los bloqueos se eliminan si la cuenta asociada deja de existir. \textbf{RF:} RF4. \textbf{RD(s):} RDW4.1. \\ 
\hspace*{1cm} \textit{Descripción:} Cuando un usuario elimina su cuenta o es eliminado, los bloqueos asociados se eliminan automáticamente, al igual que si el usuario bloqueado borra*


\section{Quinto requisito funcional}
\textbf{\textcolor{rojo}{\underline{RF4.5:}}} Añadir Amigo \\[6pt]

\textbf{\textcolor{rojo}{\underline{Descripción:}}} Permite a un usuario añadir a otro usuario como amigo vía búsqueda con el nombre de dicho usuario y su id. \\[6pt]

\textbf{\textcolor{rojo}{\underline{Entrada:}}} Agente externo: usuario. Acción: Añadir como amigo a un usuario.
Requisito de datos de entrada \textcolor{azul}{RDE4.5}. \\[6pt]

\textbf{\textcolor{rojo}{\underline{BD:}}} Requisito de datos de escritura \textcolor{azul}{RDW4.5}.\\[6pt]

\textbf{\textcolor{rojo}{\underline{Salida:}}} Agente externo: usuario. Acción: confirmación resultado.
Requisito de datos de salida: ninguno.\\[12pt]


\textcolor{azul}{\textbf{RDE4.5:}} Datos de entrada de añadir amigo. \\[4pt]
\hspace*{1cm} ID de usuario emisor: Cadena de caracteres (40)\\
\hspace*{1cm} ID de usuario a añadir como amigo: Cadena de caracteres (40)\\
\hspace*{1cm} Fecha de amistad: Fecha\\[8pt]

\textcolor{azul}{\textbf{RDW4.5:}} Datos sobre el usuario añadido como amigo. \\[4pt]
\hspace*{1cm} Se añade el ID del usuario añadido a la lista de amigos del emisor.\\
\hspace*{1cm} Estado de relación: Cadena de caracteres (15) — (“amigo”)\\
\hspace*{1cm} Fecha de añadido: Fecha\\[8pt]

\textbf{\textcolor{verde}{Restricciones Semánticas:}} \\[6pt]

\textbf{\textcolor{verde}{\underline{RS4.1:}}} Un usuario no puede añadirse a sí mismo como amigo. \textbf{RF:} RF4.5\textbf{RD(s):} RDW4.5 \\ 
\hspace*{1cm} \textit{Descripción:} El sistema debe impedir que un usuario intente agregarse a sí mismo como amigo, mostrando un mensaje de error si lo hace. \\[4pt]

\textbf{\textcolor{verde}{\underline{RS4.2:}}} No se puede añadir a un usuario inexistente o eliminado. \textbf{RF:} RF4.5 \textbf{RD(s):} RDW4.5 \\ 
\hspace*{1cm} \textit{Descripción:} Si el usuario objetivo no existe o su cuenta fue eliminada, la solicitud de amistad es rechazada. \\[4pt]

\textbf{\textcolor{verde}{\underline{RS4.3:}}} No se puede volver a añadir a un usuario que ya es amigo. \textbf{RF:} RF4.5 \textbf{RD(s):} RDW4.5 \\ 
\hspace*{1cm} \textit{Descripción:} El sistema verifica si el usuario ya se encuentra en la lista de amigos y, de ser así, impide duplicar la solicitud. \\[4pt]

\textbf{\textcolor{verde}{\underline{RS4.4:}}} Las amistades se eliminan si la cuenta asociada deja de existir. \textbf{RF:} RF4.5 \textbf{RD(s):} RDW4.5 \\ 
\hspace*{1cm} \textit{Descripción:} Cuando un usuario elimina su cuenta o es eliminado, las relaciones de amistad asociadas se eliminan automáticamente.
