\section{Ejemplo de requisito funcional}
\textbf{\textcolor{rojo}{\underline{RF1:}}} Dar de alta contacto. \\[6pt]

\textbf{\textcolor{rojo}{Entrada:}} Agente externo: usuario. Acción: solicitar inserción. 
Requisito de datos de entrada \textcolor{azul}{RDE1}. \\[6pt]

\textbf{\textcolor{rojo}{BD:}} Requisito de datos de escritura \textcolor{azul}{RDW1}. \\[6pt]

\textbf{\textcolor{rojo}{Salida:}} Agente externo: usuario. Acción: confirmación resultado. 
Requisito de datos de salida: ninguno. \\[12pt]

\textcolor{azul}{\textbf{RDE1:}} Datos de entrada de alta de contacto \\
\hspace*{1cm} Nombre: Cadena de caracteres (20) \\
\hspace*{1cm} Apellidos: Cadena de caracteres (40) \\
\hspace*{1cm} Teléfono: Cadena de caracteres (20) \\[12pt]

\textcolor{azul}{\textbf{RDW1:}} Datos almacenados de contacto. \\
\hspace*{1cm} Los mismos datos que RDE1.
