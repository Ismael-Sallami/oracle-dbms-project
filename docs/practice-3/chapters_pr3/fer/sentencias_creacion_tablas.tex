\section{Sentencias de Creación de Tablas del Subsistema de Usuarios (Fernando)}

En este subsistema se definen las estructuras relacionales necesarias para gestionar la identidad de los usuarios y sus relaciones sociales básicas dentro de la plataforma. Las sentencias DDL se han diseñado aplicando restricciones de integridad (\textit{Primary Key}, \textit{Foreign Key}, \textit{Unique} y \textit{Check}) con el fin de garantizar consistencia en el modelo y evitar registros inválidos.

\subsection{Tabla \texttt{USUARIO}}
\textbf{Descripción:} La tabla utiliza un identificador autogenerado (\texttt{IDENTITY}) como clave primaria. Se definen restricciones \texttt{UNIQUE} para \texttt{EMAIL} y \texttt{NOMBREUSUARIO}, y restricciones \texttt{CHECK} para los campos de control (\texttt{ESADMIN} y \texttt{BORRADO}).

\begin{lstlisting}[language=SQL, caption={Creación de la tabla USUARIO}]
CREATE TABLE USUARIO (
    IDUSUARIO           NUMBER GENERATED BY DEFAULT AS IDENTITY,
    NOMBREUSUARIO       VARCHAR2(20) NOT NULL,
    EMAIL               VARCHAR2(50) NOT NULL,
    CONTRASENIA         VARCHAR2(128) NOT NULL,
    IMAGENDEPERFIL      VARCHAR2(255),
    BIOGRAFIA           VARCHAR2(100),
    FECHAMODIFICACION   DATE,
    FECHAELIMINACION    DATE,
    ESADMIN             NUMBER DEFAULT 0 CHECK (ESADMIN IN (0, 1)),
    BORRADO             VARCHAR2(1) DEFAULT 'N' CHECK (BORRADO IN ('Y', 'N')),
    
    CONSTRAINT PK_USUARIO PRIMARY KEY (IDUSUARIO),
    CONSTRAINT UK_USUARIO_EMAIL UNIQUE (EMAIL),
    CONSTRAINT UK_USUARIO_NOMBRE UNIQUE (NOMBREUSUARIO)
);
\end{lstlisting}

\subsection{Tabla \texttt{AMISTAD}}

\textbf{Descripción:} Se utiliza una clave primaria compuesta (\texttt{IDUSUARIO1, IDUSUARIO2}) y dos claves ajenas contra \texttt{USUARIO}. Se aplica \texttt{ON DELETE CASCADE} para eliminar automáticamente relaciones cuando un usuario es eliminado. Además, se incluye una restricción \texttt{CHECK} para impedir auto-relaciones.

\begin{lstlisting}[language=SQL, caption={Creación de la tabla AMISTAD}]
CREATE TABLE AMISTAD (
    IDUSUARIO1          NUMBER NOT NULL,
    IDUSUARIO2          NUMBER NOT NULL,
    FECHAAMISTAD        DATE DEFAULT SYSDATE NOT NULL,
    
    CONSTRAINT PK_AMISTAD PRIMARY KEY (IDUSUARIO1, IDUSUARIO2),
    CONSTRAINT FK_AMIST_U1 FOREIGN KEY (IDUSUARIO1) REFERENCES USUARIO(IDUSUARIO) ON DELETE CASCADE,
    CONSTRAINT FK_AMIST_U2 FOREIGN KEY (IDUSUARIO2) REFERENCES USUARIO(IDUSUARIO) ON DELETE CASCADE,
    CONSTRAINT CK_AMIST_DISTINTOS CHECK (IDUSUARIO1 <> IDUSUARIO2)
);
\end{lstlisting}

\subsection{Tabla \texttt{BLOQUEO}}

\textbf{Descripción:} Al igual que en \texttt{AMISTAD}, se implementa una clave primaria compuesta y claves ajenas hacia \texttt{USUARIO}. Se impide la auto-relación mediante \texttt{CHECK} y se asegura eliminación en cascada.

\begin{lstlisting}[language=SQL, caption={Creación de la tabla BLOQUEO}]
CREATE TABLE BLOQUEO (
    IDUSUARIO1          NUMBER NOT NULL,
    IDUSUARIO2          NUMBER NOT NULL,
    FECHABLOQUEO        DATE DEFAULT SYSDATE NOT NULL,
    
    CONSTRAINT PK_BLOQUEO PRIMARY KEY (IDUSUARIO1, IDUSUARIO2),
    CONSTRAINT FK_BLOQ_U1 FOREIGN KEY (IDUSUARIO1) REFERENCES USUARIO(IDUSUARIO) ON DELETE CASCADE,
    CONSTRAINT FK_BLOQ_U2 FOREIGN KEY (IDUSUARIO2) REFERENCES USUARIO(IDUSUARIO) ON DELETE CASCADE,
    CONSTRAINT CK_BLOQ_DISTINTOS CHECK (IDUSUARIO1 <> IDUSUARIO2)
);
\end{lstlisting}
