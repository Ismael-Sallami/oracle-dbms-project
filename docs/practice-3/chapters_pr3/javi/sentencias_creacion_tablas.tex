\section{Sentencias DDL del Subsistema de Publicaciones (Javi)}

El diseño físico del subsistema de Publicaciones se centra en la eficiencia de las consultas de contenido y la integridad de las interacciones sociales (likes). A continuación se detallan las sentencias DDL utilizadas.

\subsection{Tabla: PUBLICACION}
Esta tabla almacena el contenido generado por los usuarios. Se ha implementado una columna \texttt{ELIMINADO} para permitir el borrado lógico, preservando la integridad referencial histórica.

\begin{lstlisting}[language=SQL, caption={Creación de la tabla PUBLICACION}]
CREATE TABLE PUBLICACION (
    IDPUBLICACION     NUMBER GENERATED BY DEFAULT AS IDENTITY,
    NOMBRE            VARCHAR2(30) NOT NULL,
    DESCRIPCION       VARCHAR2(256),
    CATEGORIA         VARCHAR2(32),
    IMAGEN            VARCHAR2(255),
    FECHACREACION     DATE DEFAULT SYSDATE,
    FECHAMODIFICACION DATE,
    ELIMINADO         VARCHAR2(1) DEFAULT 'N' CHECK (ELIMINADO IN ('Y', 'N')),
    IDUSUARIO         NUMBER NOT NULL,
    
    -- IDPUBLICACION como clave primara
    CONSTRAINT PK_PUBLICACION PRIMARY KEY (IDPUBLICACION),
    -- Integridad referencial con el autor
    CONSTRAINT FK_PUB_USUARIO FOREIGN KEY (IDUSUARIO)
        REFERENCES USUARIO(IDUSUARIO) ON DELETE CASCADE
);
\end{lstlisting}

\textbf{Justificación de diseño:}
\begin{itemize}
    \item \textbf{Borrado Lógico (ELIMINADO):} En lugar de eliminar físicamente los registros, se marcan con un flag. Esto es crucial para mantener la trazabilidad y evitar errores en cascada si existieran reportes asociados.
    \item \textbf{Restricciones de Tamaño:} Se han limitado los campos \texttt{TITULO} y \texttt{DESCRIPCION} acorde a los requisitos de la interfaz de usuario para garantizar una visualización uniforme en el feed.
\end{itemize}

\subsection{Tabla: ME\_GUSTA}
Tabla relacional que implementa la relación N:M entre usuarios y publicaciones para gestionar las interacciones.

\begin{lstlisting}[language=SQL, caption={Creación de la tabla ME\_GUSTA}]
CREATE TABLE ME_GUSTA (
    IDUSUARIO     NUMBER NOT NULL,
    IDPUBLICACION NUMBER NOT NULL,
    FECHA_LIKE    DATE DEFAULT SYSDATE,
    
    CONSTRAINT PK_ME_GUSTA PRIMARY KEY (IDUSUARIO, IDPUBLICACION),
    
    CONSTRAINT FK_LIKE_USUARIO FOREIGN KEY (IDUSUARIO)
        REFERENCES USUARIO(IDUSUARIO) ON DELETE CASCADE,
    CONSTRAINT FK_LIKE_PUB FOREIGN KEY (IDPUBLICACION)
        REFERENCES PUBLICACION(IDPUBLICACION) ON DELETE CASCADE
);

-- Índices para optimizar el conteo de likes y la verificación de estado
CREATE INDEX IDX_ME_GUSTA_PUB ON ME_GUSTA (IDPUBLICACION);
CREATE INDEX IDX_ME_GUSTA_USR_PUB ON ME_GUSTA (IDUSUARIO, IDPUBLICACION);
\end{lstlisting}
